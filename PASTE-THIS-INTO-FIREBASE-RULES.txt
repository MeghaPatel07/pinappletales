// ============================================================================
// STEP 2 of 2 — PERMANENT RULES
//
// Your admin record now exists, so the temporary write permission is no longer
// needed. Paste this whole file into the Firebase Rules tab and click Publish.
//
// This locks the admins collection so nobody can grant themselves admin access
// from a browser. Everything else is unchanged, and this does NOT affect your
// ability to log in.
//
// This is identical to firestore.rules in the project.
// ============================================================================

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid));
    }

    function publiclyReadable() {
      return resource.data.isActive == true;
    }

    match /blogs/{blogId} {
      allow read: if publiclyReadable() || isAdmin();
      allow write: if isAdmin();
    }

    match /events/{eventId} {
      allow read: if publiclyReadable() || isAdmin();
      allow write: if isAdmin();
    }

    match /podcasts/{podcastId} {
      allow read: if publiclyReadable() || isAdmin();
      allow write: if isAdmin();
    }

    match /eventForms/{eventId} {
      allow read: if publiclyReadable() || isAdmin();
      allow write: if isAdmin();
    }

    match /eventRegistrations/{registrationId} {
      allow read, update, delete: if isAdmin();
      allow create: if isValidRegistration();
    }

    function isValidRegistration() {
      let data = request.resource.data;
      return data.keys().hasAll(['eventId', 'values', 'createdAt'])
        && data.eventId is string
        && data.eventId.size() > 0
        && data.eventId.size() <= 128
        && data.values is map
        && data.values.size() <= 40
        && data.createdAt == request.time
        && request.resource.size() < 20000;
    }

    // Locked: readable by the admin it belongs to, never writable from a client.
    match /admins/{uid} {
      allow read: if request.auth != null && request.auth.uid == uid;
      allow write: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
